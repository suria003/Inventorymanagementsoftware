UsersTable = CREATE TABLE users ( id, INT AUTO_INCREMENT NOT NULL, name VARACHAR(50) NOT NULL, username VARCHAR(50) PRIMARY KEY UNIQUE NOT NULL, location VARCHAR(20) NOT NULL, user_Token VARCHAR(20) UNIQUE NOT NULL, product_Id VARACHAR(20) UNIQUE NOT NULL);
ProductTable = CREATE TABLE product (id INT AUTO_INCREMENT PRIMARY KEY UNIQUE NOT NULL, product_Name VARCHAR(50) NOT NULL, product_Id VARCHAR(20) NOT NULL, product_Price FLOAT NOT NULL, product_Quantity BIGINT NOT NULL);
OrdersTable = CREATE TABLE orders ( id INT AUTO_INCREMENT UNIQUE NOT NULL, order_Id VARCHAR(20) PRIMARY KEY UNIQUE NOT NULL, user_Token VARCHAR(20) NOT NULL, product_Name VARCHAR(50) NOT NULL, product_Price FLOAT NOT NULL, product_Qty BIGINT NOT NULL, ordered_By TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL, status VARCHAR(10) NOT NULL, product_Id VARCAHR(20) NOT NULL);
ProductMovementsTable = CREATE TABLE productMovement ( id INT AUTO_INCREMENT UNIQUE NOT NULL, movement_Id VARCHAR(20) PRIMARY KEY UNIQUE ID NOT NULL, order_Id VARCHAR(20) NOT NULL, from_Location VARCHAR(50) NOT NULL, to_Location VARCHAR(50) NOT NULL, delivery TIMESTAMP NOT NULL);
